"use client";

import Link from "next/link";
import { useEffect, useState } from "react";

type AccountUser = {
  id: string;
  name: string;
  email: string;
  company: string | null;
  phone: string | null;
  plan: string;
};

type AccountResponse = {
  success?: boolean;
  message?: string;
  error?: string;
  user?: AccountUser;
};

type Notice = {
  type: "success" | "error";
  text: string;
} | null;

function getPlanLabel(plan: string): string {
  const normalized = plan.trim().toLowerCase();

  if (normalized === "agency") {
    return "Agency";
  }

  if (normalized === "pro") {
    return "Pro";
  }

  if (
    normalized === "founder" ||
    normalized === "standard"
  ) {
    return "Founder";
  }

  return "Free";
}

    async function loadAccount() {
      try {
        setLoading(true);
        setNotice(null);

        const response = await fetch("/api/account", {
          method: "GET",
          cache: "no-store",
          credentials: "include",
          signal: controller.signal,
        });

        const data = (await response.json().catch(() => null)) as
          | AccountResponse
          | null;

        if (!response.ok || !data?.user) {
          throw new Error(
            data?.error || "Die Kontodaten konnten nicht geladen werden."
          );
        }

        setName(data.user.name || "");
        setEmail(data.user.email || "");
        setCompany(data.user.company || "");
        setPhone(data.user.phone || "");
        setPlan(data.user.plan || "free");

        if (
          error instanceof DOMException &&
          error.name === "AbortError"
        ) {
          return;
        }

        setNotice({
          type: "error",
          text:
            error instanceof Error
              ? error.message
              : "Die Kontodaten konnten nicht geladen werden.",
        });
      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    }

    void loadAccount();

    return () => {
      controller.abort();

            credentials: "include",
            cache: "no-store",
            headers: {

                "application/json",

            body: JSON.stringify({

            }),
          }
        );

          .json()
          .catch(() => null)) as
          | {
              success?: boolean;
              message?: string;
              error?: string;
            }
          | null;

          !response.ok ||
          !data?.success

            data?.error ||
              "Das Founder-Abonnement konnte nicht bestätigt werden."
          );

         * Navbar, Sessionanzeige und Konto laden

         */

        );

        sessionStorage.removeItem(
          verificationKey
        );

          return;
        }

        setNotice({
          type: "error",
          text:
            error instanceof Error
              ? error.message
              : "Das Founder-Abonnement konnte nicht bestätigt werden.",
        });
      }

    return () => {
      cancelled = true;
    };

          company,

      localStorage.setItem("companyName", data.user.company || "");
      localStorage.setItem("userPhone", data.user.phone || "");

        type: "success",
        text: data.message || "Kontaktdaten wurden gespeichert.",

        type: "error",
        text:
          error instanceof Error
            ? error.message
            : "Die Kontodaten konnten nicht gespeichert werden.",

      setSaving(false);
